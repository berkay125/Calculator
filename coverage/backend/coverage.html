// Command server serves the calculator API and the frontend.
package main

import (
        "context"
        "errors"
        "flag"
        "fmt"
        "io"
        "io/fs"
        "log/slog"
        "net"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "calculator/internal/api"
        "calculator/internal/web"
)

const (
        defaultAddr = ":8080"

        readHeaderTimeout = 5 * time.Second
        readTimeout       = 10 * time.Second
        writeTimeout      = 10 * time.Second
        idleTimeout       = 60 * time.Second
        // shutdownTimeout is how long requests in flight get to finish. docker
        // stop kills the process 10 seconds after it sends SIGTERM.
        shutdownTimeout = 5 * time.Second
)

// shutdownSignals stop the server gracefully: Ctrl+C sends os.Interrupt, and
// docker stop sends SIGTERM.
var shutdownSignals = []os.Signal{os.Interrupt, syscall.SIGTERM}

func main() {
        ctx, stop := signal.NotifyContext(context.Background(), shutdownSignals...)
        err := run(ctx, os.Args[1:], os.Getenv, os.Stderr)
        stop()
        if err != nil {
                fmt.Fprintln(os.Stderr, "server:", err)
                os.Exit(1)
        }
}

// run serves the app until ctx is cancelled, then shuts down gracefully and
// returns nil. args are the command-line arguments without the program name,
// getenv reads environment variables, and logs go to stderr as JSON.
func run(ctx context.Context, args []string, getenv func(string) string, stderr io.Writer) error {
        cfg, err := parseConfig(args, getenv, stderr)
        if errors.Is(err, flag.ErrHelp) {
                return nil
        }
        if err != nil {
                return err
        }
        log := slog.New(slog.NewJSONHandler(stderr, nil))
        handler, err := newMux(log, web.Dist())
        if err != nil {
                return err
        }

        ln, err := net.Listen("tcp", cfg.addr)
        if err != nil {
                return err
        }
        srv := &http.Server{
                Handler:           handler,
                ReadHeaderTimeout: readHeaderTimeout,
                ReadTimeout:       readTimeout,
                WriteTimeout:      writeTimeout,
                IdleTimeout:       idleTimeout,
                ErrorLog:          slog.NewLogLogger(log.Handler(), slog.LevelError),
        }
        log.Info("listening", "addr", ln.Addr().String())

        serveErr := make(chan error, 1)
        go func() { serveErr <- srv.Serve(ln) }()
        select {
        case err := <-serveErr:
                // Shutdown has not been called, so Serve failed.
                return err
        case <-ctx.Done():
        }

        log.Info("shutting down")
        shutdownCtx, cancel := context.WithTimeout(context.WithoutCancel(ctx), shutdownTimeout)
        defer cancel()
        if err := srv.Shutdown(shutdownCtx); err != nil {
                return fmt.Errorf("shutdown: %w", err)
        }
        log.Info("stopped")
        return nil
}

// newMux serves the API under /api/ and the frontend build static at /. The
// API has its own mux, so it answers every /api/ path itself:
// GET /api/v1/calculate gets 405 and never falls through to the frontend.
func newMux(log *slog.Logger, static fs.FS) (http.Handler, error) {
        files, err := web.NewHandler(static)
        if err != nil {
                return nil, err
        }
        if _, err := fs.Stat(static, "index.html"); err != nil {
                log.Warn("the frontend is not built into the binary; serving only the API")
        }
        mux := http.NewServeMux()
        mux.Handle("/api/", api.NewHandler(log))
        mux.Handle("/", files)
        return mux, nil
}

type config struct {
        addr string
}

// parseConfig reads the listen address from the -addr flag, then the ADDR
// environment variable, then defaultAddr. Usage and flag errors are written
// to stderr; -h returns flag.ErrHelp.
func parseConfig(args []string, getenv func(string) string, stderr io.Writer) (config, error) {
        addr := getenv("ADDR")
        if addr == "" {
                addr = defaultAddr
        }
        var cfg config
        fs := flag.NewFlagSet("server", flag.ContinueOnError)
        fs.SetOutput(stderr)
        fs.StringVar(&cfg.addr, "addr", addr, "listen `address`, such as :8080 (env ADDR)")
        if err := fs.Parse(args); err != nil {
                return config{}, err
        }
        if fs.NArg() > 0 {
                return config{}, fmt.Errorf("unexpected arguments: %q", fs.Args())
        }
        // An empty address would listen on a random port.
        if cfg.addr == "" {
                return config{}, errors.New("-addr must not be empty")
        }
        return cfg, nil
}

import (

        "errors"
        "net/http"
)

// The codes of the errors the handler finds itself.
const (
        codeInvalidRequest    = "invalid_request"
        codeInvalidAns        = "invalid_ans"
        codeExpressionTooLong = "expression_too_long"
)

// statusByCode maps every error code to its HTTP status: 400 for invalid
// input, 422 for undefined math. A code missing here is answered with 500.
var statusByCode = map[string]int{
        codeInvalidRequest:     http.StatusBadRequest,
        codeInvalidAns:         http.StatusBadRequest,
        codeExpressionTooLong:  http.StatusBadRequest,
        "nesting_too_deep":     http.StatusBadRequest,
        "syntax_error":         http.StatusBadRequest,
        "unknown_function":     http.StatusBadRequest,
        "unknown_variable":     http.StatusBadRequest,
        "wrong_argument_count": http.StatusBadRequest,
        "division_by_zero":     http.StatusUnprocessableEntity,
        "domain_error":         http.StatusUnprocessableEntity,
        "overflow":             http.StatusUnprocessableEntity,
        "underflow":            http.StatusUnprocessableEntity,
}

// coded is an error with an error code. The handler knows errors only through
// this interface and positional, not through their concrete types.
type coded interface {
        error
        Code() string
}

// positional is an error that also knows where in the expression it is.
type positional interface {
        Pos() int
}

type errorResponse struct {
        Error errorBody `json:"error"`
}

type errorBody struct {
        Code    string `json:"code"`
        Message string `json:"message"`
        // Position is a pointer so that 0 is written and a missing position is
        // left out.
        Position *int `json:"position,omitempty"`
}

// writeError writes an error with a code from statusByCode.
func writeError(w http.ResponseWriter, code, message string, position *int) {
        body := errorBody{Code: code, Message: message, Position: position}
        writeJSON(w, statusByCode[code], errorResponse{body})
}

// writeEvalError answers an error from evaluating expression. An error without
// a known code is a bug: it is logged, and the client gets 500.
func (h *handler) writeEvalError(w http.ResponseWriter, expression string, err error) {
        var c coded
        if errors.As(err, &c) {
                if _, ok := statusByCode[c.Code()]; ok {
                        var position *int
                        if p, ok := c.(positional); ok {
                                pos := p.Pos()
                                position = &pos
                        }
                        writeError(w, c.Code(), c.Error(), position)
                        return
                }
        }
        h.log.Error("evaluate failed", "expression", expression, "error", err)
        body := errorBody{Code: "internal_error", Message: "internal server error"}
        writeJSON(w, http.StatusInternalServerError, errorResponse{body})
}

func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        // An error here means the client has gone away; there is no one to tell.
        _ = json.MarshalWrite(w, v)
}

package api

import (
        "encoding/json/jsontext"
        "encoding/json/v2"
        "errors"
        "fmt"
        "io"
        "log/slog"
        "net/http"

        "calculator/internal/eval"

// Input limits. The endpoint is public, so every input is bounded.
const (
        maxBodyBytes = 4 << 10
        // maxExpressionLength counts bytes. Only ASCII is valid in an expression,
        // so for valid input that is the number of characters.
        maxExpressionLength = 1000
)

type request struct {
        // Pointers tell a missing field from an empty one: a missing expression is
        // an invalid request, an empty one is a syntax error.
        Expression *string `json:"expression"`
        Ans        *string `json:"ans"`
}

type response struct {
        Result string `json:"result"`
}

// evaluateFunc is the signature of eval.Evaluate. Tests replace it to reach
// errors that eval never returns.
type evaluateFunc func(expression string, vars map[string]eval.Value) (eval.Value, error)

type handler struct {
        evaluate evaluateFunc
        log      *slog.Logger
}

// NewHandler returns the API's handler, which serves POST /api/v1/calculate.
// Other methods get 405. Unexpected errors are logged to log and answered
// with 500.
func NewHandler(log *slog.Logger) http.Handler {
        return newHandler(eval.Evaluate, log)
}

func newHandler(evaluate evaluateFunc, log *slog.Logger) http.Handler {
        h := &handler{evaluate: evaluate, log: log}
        mux := http.NewServeMux()
        mux.HandleFunc("POST /api/v1/calculate", h.calculate)
        return mux
}

func (h *handler) calculate(w http.ResponseWriter, r *http.Request) {
        req, err := decodeRequest(w, r)
        if err != nil {
                writeError(w, codeInvalidRequest, err.Error(), nil)
                return
        }
        expression := *req.Expression
        if len(expression) > maxExpressionLength {
                msg := fmt.Sprintf("expression is longer than %d characters", maxExpressionLength)
                writeError(w, codeExpressionTooLong, msg, nil)
                return
        }
        var vars map[string]eval.Value
        if req.Ans != nil {
                // Every ParseValue error is invalid input, including overflow, so its
                // code does not go through the status table.
                ans, err := eval.ParseValue(*req.Ans)
                if err != nil {
                        writeError(w, codeInvalidAns, ansMessage(err), nil)
                        return
                }
                vars = map[string]eval.Value{"Ans": ans}
        }
        result, err := h.evaluate(expression, vars)
        if err != nil {
                h.writeEvalError(w, expression, err)
                return
        }
        writeJSON(w, http.StatusOK, response{Result: result.String()})
}

// decodeRequest reads the JSON body: exactly one object whose member names
// match exactly. An unknown, differently cased or repeated name is an error,
// because ignoring a misspelled "ans" would silently give a wrong result. A
// returned error's message is meant for the client.
func decodeRequest(w http.ResponseWriter, r *http.Request) (request, error) {
        data, err := io.ReadAll(http.MaxBytesReader(w, r.Body, maxBodyBytes))
        if _, ok := errors.AsType[*http.MaxBytesError](err); ok {
                return request{}, fmt.Errorf("request body is larger than %d bytes", maxBodyBytes)
        }
        if err != nil {
                return request{}, errors.New("request body could not be read")
        }
        if len(data) == 0 {
                return request{}, errors.New("request body is empty")
        }
        var req request
        if err := json.Unmarshal(data, &req, json.RejectUnknownMembers(true)); err != nil {
                return request{}, decodeError(err)
        }
        if req.Expression == nil {
                return request{}, errors.New("expression is missing")
        }
        return req, nil
}

// decodeError describes a JSON decoding error without Go type names.
func decodeError(err error) error {
        if serr, ok := errors.AsType[*jsontext.SyntacticError](err); ok && serr.Err == jsontext.ErrDuplicateName {
                return fmt.Errorf("duplicate field %q", serr.JSONPointer.LastToken())
        }
        serr, ok := errors.AsType[*json.SemanticError](err)
        switch {
        case !ok:
                return errors.New("request body is not valid JSON")
        case serr.Err == json.ErrUnknownName:
                return fmt.Errorf("unknown field %q", serr.JSONPointer.LastToken())
        case serr.JSONPointer == "":
                return errors.New("request body must be a JSON object")
        }
        return fmt.Errorf("%s must be a string", serr.JSONPointer.LastToken())
}

// ansMessage describes why ans was rejected.
func ansMessage(err error) string {
        var outOfRange coded // overflow or underflow
        switch {
        case errors.Is(err, eval.ErrTooManyDigits):
                return "invalid ans: " + err.Error()
        case errors.As(err, &outOfRange):
                return "invalid ans: outside the exponent range"
        }
        return "invalid ans: not a valid number"
}

package arith

import (
        "errors"
        "fmt"

        "github.com/cockroachdb/apd/v3"
)

// Precision is the number of significant digits every operation rounds its
// result to. It matches IEEE 754 decimal128.
const Precision = 34

// The exponent range also matches decimal128. Results outside it are reported
// as overflow or underflow instead of growing without bound.
const (
        maxExponent = 6144
        minExponent = -6143
)

// ctx is shared by all operations. An apd context is safe for concurrent use
// as long as it is not modified. Ties round to even, the IEEE 754 default.
var ctx = &apd.Context{
        Precision:   Precision,
        MaxExponent: maxExponent,
        MinExponent: minExponent,
        Traps:       apd.DefaultTraps,
        Rounding:    apd.RoundHalfEven,
}

// parseCtx keeps every digit of its input (precision 0 disables rounding) but
// still enforces the exponent range. Rounding inputs before an operation would
// make results depend on digits that were already thrown away.
//
// It sets no traps: when a trapped parse fails, apd drops the condition flags,
// and Parse needs them to tell overflow from underflow.
var parseCtx = &apd.Context{
        MaxExponent: maxExponent,
        MinExponent: minExponent,
        Rounding:    apd.RoundHalfEven,
}

// ErrInvalidNumber is returned by Parse for strings that are not finite
// decimal numbers.
var ErrInvalidNumber = errors.New("invalid number")

// Parse converts a decimal string such as "12.5", "-3" or "1.5E+40" into a
// Decimal. All digits are kept; only operation results are rounded. A
// malformed string returns ErrInvalidNumber, and a number outside the exponent
// range returns a MathError of kind Overflow or Underflow.
func Parse(s string) (*apd.Decimal, error) {
        d, cond, err := parseCtx.NewFromString(s)
        if err == nil && cond&(apd.Overflow|apd.Underflow|apd.Subnormal) != 0 {
                return nil, fromCondition(cond, nil)
        }
        if err != nil || d.Form != apd.Finite {
                return nil, fmt.Errorf("%w: %q", ErrInvalidNumber, s)
        }
        return d, nil
}

// Neg returns -x. Negation is exact, so unlike the other operations it does
// not round, and it never returns negative zero.
func Neg(x *apd.Decimal) *apd.Decimal {
        return new(apd.Decimal).Neg(x)
}

// Round rounds x to Precision significant digits, ties to even. Operation
// results are already rounded; Round is for values that skipped every
// operation, such as a long literal. Rounding up can overflow.
func Round(x *apd.Decimal) (*apd.Decimal, error) {
        d := new(apd.Decimal)
        if cond, err := ctx.Round(d, x); err != nil {
                return nil, fromCondition(cond, err)
        }
        return d, nil
}

// Add returns x + y.
func Add(x, y *apd.Decimal) (*apd.Decimal, error) {
        return apply(ctx.Add, x, y)
}

// Sub returns x - y.
func Sub(x, y *apd.Decimal) (*apd.Decimal, error) {
        return apply(ctx.Sub, x, y)
}

// Mul returns x * y.
func Mul(x, y *apd.Decimal) (*apd.Decimal, error) {
        return apply(ctx.Mul, x, y)
}

// Div returns x / y. Any division by zero, including 0/0, is reported as
// DivisionByZero.
func Div(x, y *apd.Decimal) (*apd.Decimal, error) {
        return apply(ctx.Quo, x, y)
}

// Pow returns x raised to the power y. 0^0 is 1 and zero raised to a negative
// power is DivisionByZero. A negative x with a non-integer y is a DomainError.
// Non-integer powers are computed through ln/exp and are approximate.
func Pow(x, y *apd.Decimal) (*apd.Decimal, error) {
        if x.IsZero() {
                switch y.Sign() {
                case 0:
                        // apd follows the GDA spec, where 0^0 is undefined.
                        return apd.New(1, 0), nil
                case -1:
                        // apd returns Infinity here without raising a condition.
                        return nil, &MathError{Kind: DivisionByZero}
                }
        }
        d, err := pow(x, y)
        var mathErr *MathError
        if errors.As(err, &mathErr) && (mathErr.Kind == Overflow || mathErr.Kind == Underflow) {
                mathErr.Kind = powRangeKind(x, y)
        }
        return d, err
}

// powCtx computes the two parts of a split power (see pow). It has ten guard
// digits and apd's full exponent range, so the parts are rounded and checked
// against the exponent range only once more, in their product.
var powCtx = &apd.Context{
        Precision:   Precision + 10,
        MaxExponent: apd.MaxExponent,
        MinExponent: apd.MinExponent,
        Traps:       apd.DefaultTraps,
        Rounding:    apd.RoundHalfEven,
}

// pow computes x^y. For a non-integer y, apd computes ln and exp at a
// precision above the number of digits in x. With a long base, such as a
// base of thousands of digits, that takes half a second, and from about 2700
// digits apd gives up with an error that has no condition. A base with more
// digits than powCtx keeps is therefore split: x^y = x^int(y) * x^frac(y).
//   - The integer power multiplies with every digit of x.
//   - The fractional power uses x rounded to powCtx precision. Since
//     |frac(y)| < 1, its relative error is below that of the rounded x.
//
// Both parts lie on the same side of 1, and x^y is further from 1 than either
// of them. A part that leaves apd's exponent range therefore means that x^y
// leaves the narrower range of ctx too.
func pow(x, y *apd.Decimal) (*apd.Decimal, error) {
        var integ, frac apd.Decimal
        y.Modf(&integ, &frac)
        if frac.IsZero() || x.NumDigits() <= int64(powCtx.Precision) {
                return apply(ctx.Pow, x, y)
        }
        // Rounding cannot fail: x lies within the exponent range of ctx, far
        // inside that of powCtx.
        var short apd.Decimal
        powCtx.Round(&short, x)
        // The fractional part goes first, so a negative base is a DomainError
        // even when the integer part alone would overflow.
        fracPow, err := apply(powCtx.Pow, &short, &frac)
        if err != nil {
                return nil, err
        }
        intPow, err := apply(powCtx.Pow, x, &integ)
        if err != nil {
                return nil, err
        }
        return Mul(intPow, fracPow)
}

var one = apd.New(1, 0)

// powRangeKind tells whether an out-of-range x^y is too large or too small.
// apd cannot be trusted here: for a negative exponent it computes the positive
// power first and only turns overflow into underflow, never the reverse, so
// 0.5^-1E+100 comes back as underflow. |x|^y grows without bound when |x| > 1
// and y > 0, or when |x| < 1 and y < 0.
func powRangeKind(x, y *apd.Decimal) Kind {
        var abs apd.Decimal
        abs.Abs(x)
        if (abs.Cmp(one) > 0) == (y.Sign() > 0) {
                return Overflow
        }
        return Underflow
}

// Sqrt returns the square root of x. A negative x is a DomainError.
func Sqrt(x *apd.Decimal) (*apd.Decimal, error) {
        d := new(apd.Decimal)
        if cond, err := ctx.Sqrt(d, x); err != nil {
                return nil, fromCondition(cond, err)
        }
        return d, nil
}

var hundred = apd.New(100, 0)

// Percent returns x / 100, the value of "x%" on its own or after * and /.
func Percent(x *apd.Decimal) (*apd.Decimal, error) {
        return Div(x, hundred)
}

// PercentOf returns pct percent of base (base * pct / 100). It gives "a + b%"
// its phone-calculator meaning: a + PercentOf(a, b).
func PercentOf(base, pct *apd.Decimal) (*apd.Decimal, error) {
        p, err := Mul(base, pct)
        if err != nil {
                return nil, err
        }
        return Div(p, hundred)
}

// apply runs a binary apd operation into a new Decimal and translates a
// trapped condition into a MathError.
func apply(op func(d, x, y *apd.Decimal) (apd.Condition, error), x, y *apd.Decimal) (*apd.Decimal, error) {
        d := new(apd.Decimal)
        if cond, err := op(d, x, y); err != nil {
                return nil, fromCondition(cond, err)
        }
        return d, nil
}

import (
        "fmt"

        "github.com/cockroachdb/apd/v3"
)

// Kind classifies a MathError. Each kind has a stable error code that clients
// can rely on.
type Kind int

const (
        DivisionByZero Kind = iota + 1
        DomainError
        Overflow
        Underflow
)

var kinds = [...]struct{ code, message string }{
        DivisionByZero: {"division_by_zero", "division by zero"},
        DomainError:    {"domain_error", "result is not a real number"},
        Overflow:       {"overflow", "result is too large"},
        Underflow:      {"underflow", "result is too close to zero"},
}

// Code returns the machine-readable error code of k, such as
// "division_by_zero".
func (k Kind) Code() string {
        return kinds[k].code
}

// MathError reports a result that is mathematically undefined or cannot be
// represented.
type MathError struct {
        Kind Kind
}

func (e *MathError) Error() string {
        return kinds[e.Kind].message
}

// Code returns the error code of e's kind.
func (e *MathError) Code() string {
        return e.Kind.Code()
}

// fromCondition translates the flags of a trapped apd operation into a
// MathError. Errors without a known flag are wrapped and returned as is.
func fromCondition(cond apd.Condition, err error) error {
        switch {
        case cond&(apd.DivisionByZero|apd.DivisionUndefined) != 0:
                return &MathError{Kind: DivisionByZero}
        case cond&(apd.Overflow|apd.SystemOverflow) != 0:
                return &MathError{Kind: Overflow}
        case cond&(apd.Underflow|apd.Subnormal|apd.SystemUnderflow) != 0:
                return &MathError{Kind: Underflow}
        case cond&(apd.InvalidOperation|apd.DivisionImpossible) != 0:
                return &MathError{Kind: DomainError}
        }
        return fmt.Errorf("arith: %w", err)
}

package arith

import "github.com/cockroachdb/apd/v3"

// Results whose adjusted exponent (the exponent in scientific notation) lies
// in this range are written in plain notation, others in scientific notation.
// The upper bound lets every integer of up to Precision digits print in full.
const (
        plainMinExponent = -7
        plainMaxExponent = Precision - 1
)

// Format returns the canonical string form of d: trailing zeros removed,
// negative zero written as "0", plain notation for moderate magnitudes and
// scientific notation such as "1.5E+40" otherwise. Parse accepts every string
// Format produces.
func Format(d *apd.Decimal) string {
        var r apd.Decimal
        r.Reduce(d) // also turns -0 into 0
        adj := int64(r.Exponent) + r.NumDigits() - 1
        if adj >= plainMinExponent && adj <= plainMaxExponent {
                return r.Text('f')
        }
        return r.Text('E')
}

package arith

import "github.com/cockroachdb/apd/v3"

// Func is a function that expressions can call by name, such as sqrt.
type Func struct {
        // Arity is the exact number of arguments the function takes.
        Arity int
        // Call computes the result. Callers must pass exactly Arity arguments.
        Call func(args []*apd.Decimal) (*apd.Decimal, error)
}

// functions holds every callable function under the name used in expressions.
// Names are case-sensitive. Adding a function only needs an entry here; the
// parser accepts any name and the evaluator looks it up.
var functions = map[string]Func{
        "sqrt": {Arity: 1, Call: func(args []*apd.Decimal) (*apd.Decimal, error) {
                return Sqrt(args[0])
        }},
}

// Lookup returns the function registered under name.
func Lookup(name string) (Func, bool) {
        f, ok := functions[name]
        return f, ok
}

package eval

import (
        "errors"
        "fmt"

        "calculator/internal/arith"
        "calculator/internal/parser"
)

// Error is a problem found while evaluating a parsed expression: an unknown
// name, a wrong number of arguments, or a math error such as division by zero.
type Error struct {
        code   string
        msg    string
        offset int
        err    error // the *arith.MathError behind a math error
}

func (e *Error) Error() string {
        return e.msg
}

// Code returns the error code, such as "unknown_function" or
// "division_by_zero".
func (e *Error) Code() string {
        return e.code
}

// Pos returns the 0-based index of the name, operator or function that caused
// the error.
func (e *Error) Pos() int {
        return e.offset
}

// Unwrap returns the *arith.MathError behind a math error, or nil.
func (e *Error) Unwrap() error {
        return e.err
}

// The codes of the errors the evaluator finds itself.
const (
        codeUnknownVariable    = "unknown_variable"
        codeUnknownFunction    = "unknown_function"
        codeWrongArgumentCount = "wrong_argument_count"
)

// Codes returns every error code Evaluate can return: the parser's, the
// evaluator's own and those of math errors. The API layer maps each one to an
// HTTP status.
func Codes() []string {
        return []string{
                (&parser.SyntaxError{}).Code(),
                (&parser.NestingError{}).Code(),
                codeUnknownVariable,
                codeUnknownFunction,
                codeWrongArgumentCount,
                arith.DivisionByZero.Code(),
                arith.DomainError.Code(),
                arith.Overflow.Code(),
                arith.Underflow.Code(),
        }
}

func unknownVariable(n *parser.Variable) error {
        return &Error{code: codeUnknownVariable, msg: fmt.Sprintf("unknown variable '%s'", n.Name), offset: n.Pos}
}

func unknownFunction(n *parser.Call) error {
        return &Error{code: codeUnknownFunction, msg: fmt.Sprintf("unknown function '%s'", n.Name), offset: n.Pos}
}

func wrongArgumentCount(n *parser.Call, want int) error {
        msg := fmt.Sprintf("wrong number of arguments for %s: want %d, got %d", n.Name, want, len(n.Args))
        return &Error{code: codeWrongArgumentCount, msg: msg, offset: n.Pos}
}

// at attaches pos to an error from arith. It returns nil when err is nil, so a
// result can be passed through as return d, at(pos, err).
func at(pos int, err error) error {
        var mathErr *arith.MathError
        if !errors.As(err, &mathErr) {
                return err // nil, or an unexpected error without a code
        }
        return &Error{code: mathErr.Code(), msg: mathErr.Error(), offset: pos, err: err}
}

package eval

import (
        "fmt"

        "github.com/cockroachdb/apd/v3"

        "calculator/internal/arith"
        "calculator/internal/parser"
)

// Evaluate parses expression and computes its value. vars holds the values of
// variables such as Ans; names are case-sensitive. The result is rounded to
// arith.Precision significant digits, even when no operation was applied (a
// long literal, or its negation).
//
// Every error has a code and the position of the input that caused it:
//   - Malformed input returns the parser's errors.
//   - Unknown names and wrong argument counts are checked before anything is
//     computed, so an invalid expression is reported as such even when
//     evaluating it would also divide by zero.
//   - Math errors point at the operator or function that failed.
func Evaluate(expression string, vars map[string]Value) (Value, error) {
        root, err := parser.Parse(expression)
        if err != nil {
                return Value{}, err
        }
        e := &evaluator{vars: vars}
        if err := e.check(root); err != nil {
                return Value{}, err
        }
        d, err := e.eval(root)
        if err != nil {
                return Value{}, err
        }
        if d, err = arith.Round(d); err != nil {
                return Value{}, at(root.Position(), err)
        }
        return Value{d}, nil
}

type evaluator struct {
        vars map[string]Value
}

// check returns the first unknown name or wrong argument count in n, in input
// order.
func (e *evaluator) check(n parser.Node) error {
        switch n := n.(type) {
        case *parser.Variable:
                if _, ok := e.vars[n.Name]; !ok {
                        return unknownVariable(n)
                }
        case *parser.Call:
                f, ok := arith.Lookup(n.Name)
                if !ok {
                        return unknownFunction(n)
                }
                if len(n.Args) != f.Arity {
                        return wrongArgumentCount(n, f.Arity)
                }
                for _, arg := range n.Args {
                        if err := e.check(arg); err != nil {
                                return err
                        }
                }
        case *parser.Unary:
                return e.check(n.X)
        case *parser.Binary:
                if err := e.check(n.L); err != nil {
                        return err
                }
                return e.check(n.R)
        case *parser.Percent:
                return e.check(n.X)
        case *parser.Group:
                return e.check(n.Inner)
        }
        return nil
}

// binaryOps maps each binary operator the parser produces to its operation.
var binaryOps = map[string]func(x, y *apd.Decimal) (*apd.Decimal, error){
        "+": arith.Add,
        "-": arith.Sub,
        "*": arith.Mul,
        "/": arith.Div,
        "^": arith.Pow,
}

// eval computes the value of n, which must have passed check. Operands are
// evaluated left to right, and the first error is returned.
func (e *evaluator) eval(n parser.Node) (*apd.Decimal, error) {
        switch n := n.(type) {
        case *parser.Number:
                d, err := arith.Parse(n.Text)
                return d, at(n.Pos, err)
        case *parser.Variable:
                return e.vars[n.Name].decimal(), nil
        case *parser.Call:
                return e.call(n)
        case *parser.Unary:
                // The only prefix operator is minus.
                x, err := e.eval(n.X)
                if err != nil {
                        return nil, err
                }
                return arith.Neg(x), nil
        case *parser.Binary:
                return e.binary(n)
        case *parser.Percent:
                x, err := e.eval(n.X)
                if err != nil {
                        return nil, err
                }
                d, err := arith.Percent(x)
                return d, at(n.Pos, err)
        case *parser.Group:
                return e.eval(n.Inner)
        }
        return nil, fmt.Errorf("eval: unexpected node %T", n)
}

func (e *evaluator) call(n *parser.Call) (*apd.Decimal, error) {
        f, _ := arith.Lookup(n.Name) // check has verified the name and the arity
        args := make([]*apd.Decimal, len(n.Args))
        for i, arg := range n.Args {
                d, err := e.eval(arg)
                if err != nil {
                        return nil, err
                }
                args[i] = d
        }
        d, err := f.Call(args)
        return d, at(n.Pos, err)
}

// binary applies a binary operator. For a + b% and a - b% it applies the
// percent rule: b% means b percent of a, so 200 + 10% is 220.
func (e *evaluator) binary(n *parser.Binary) (*apd.Decimal, error) {
        l, err := e.eval(n.L)
        if err != nil {
                return nil, err
        }
        var r *apd.Decimal
        if pct, negated, ok := percentOperand(n.R); ok && (n.Op == "+" || n.Op == "-") {
                r, err = e.percentOf(l, pct, negated)
        } else {
                r, err = e.eval(n.R)
        }
        if err != nil {
                return nil, err
        }
        d, err := binaryOps[n.Op](l, r)
        return d, at(n.Pos, err)
}

// percentOperand reports whether n is a Percent node behind any number of
// unary minus signs, and whether an odd number of them negates it. A Group is
// not looked into, which is how parentheses block the percent rule.
func percentOperand(n parser.Node) (pct *parser.Percent, negated, ok bool) {
        for {
                switch x := n.(type) {
                case *parser.Percent:
                        return x, negated, true
                case *parser.Unary:
                        negated = !negated
                        n = x.X
                default:
                        return nil, false, false
                }
        }
}

// percentOf returns b percent of a, where pct is b% and negated says whether
// minus signs in front of it flip its sign. An error points at the "%".
func (e *evaluator) percentOf(a *apd.Decimal, pct *parser.Percent, negated bool) (*apd.Decimal, error) {
        b, err := e.eval(pct.X)
        if err != nil {
                return nil, err
        }
        if negated {
                b = arith.Neg(b)
        }
        d, err := arith.PercentOf(a, b)
        return d, at(pct.Pos, err)
}

// Value is an exact decimal number: the result of Evaluate or the value of a
// variable such as Ans. The zero Value is 0.
type Value struct {
        d *apd.Decimal
}

// ErrTooManyDigits is returned by ParseValue for a number with more than
// arith.Precision digits.
var ErrTooManyDigits = fmt.Errorf("number has more than %d digits", arith.Precision)

// ParseValue parses an earlier result, such as the value of Ans, and keeps
// every digit. Plain and scientific notation are both accepted ("0.5",
// "1.5E+40"), so every string that String returns parses back to the same
// value.
//
// No result has more than arith.Precision digits, so a longer number returns
// ErrTooManyDigits instead of making every operation on it work with all its
// digits. Digits are counted on the parsed number: leading zeros do not count,
// trailing zeros do ("1.50" has three).
func ParseValue(s string) (Value, error) {
        d, err := arith.Parse(s)
        if err != nil {
                return Value{}, err
        }
        if d.NumDigits() > arith.Precision {
                return Value{}, ErrTooManyDigits
        }
        return Value{d}, nil
}

// String returns v in canonical form, as arith.Format writes it: "98",
// "0.5", "1.5E+40".
func (v Value) String() string {
        return arith.Format(v.decimal())
}

func (v Value) decimal() *apd.Decimal {
        if v.d == nil {
                return new(apd.Decimal)
        }
        return v.d
}

import (
        "fmt"
        "strings"
)

// Node is an expression in the syntax tree: a *Number, *Variable, *Call,
// *Unary, *Binary, *Percent or *Group.
type Node interface {
        // Position returns the index in the input of the token that created the
        // node, which is also its Pos field.
        Position() int
        // String prints the subtree as a fully parenthesized S-expression, such
        // as (* 2 (^ (group (+ 3 4)) 2)).
        String() string
}

// Number is a number literal, kept exactly as written.
type Number struct {
        Text string
        Pos  int
}

// Variable is a name that is not followed by "(", such as Ans. The evaluator
// resolves it.
type Variable struct {
        Name string
        Pos  int
}

// Call is a function call such as sqrt(9). The parser accepts any name; the
// evaluator looks it up and checks the number of arguments.
type Call struct {
        Name string
        Args []Node
        Pos  int // position of the name
}

// Unary is a prefix operator applied to X. The only one is "-".
type Unary struct {
        Op  string
        X   Node
        Pos int // position of the operator
}

// Binary is an infix operator: "+", "-", "*", "/" or "^".
type Binary struct {
        Op   string
        L, R Node
        Pos  int // position of the operator
}

// Percent is a postfix "%" applied to X. Its value depends on where it appears
// (200 + 10% is 220, 200 * 10% is 20), so the evaluator applies the percent
// rule, not the parser.
type Percent struct {
        X   Node
        Pos int // position of "%"
}

// Group is a parenthesized expression. It is a node of its own because
// parentheses change how percent is evaluated: 100 + (10%) is not 100 + 10%.
type Group struct {
        Inner Node
        Pos   int // position of "("
}

func (n *Number) Position() int   { return n.Pos }
func (n *Variable) Position() int { return n.Pos }
func (n *Call) Position() int     { return n.Pos }
func (n *Unary) Position() int    { return n.Pos }
func (n *Binary) Position() int   { return n.Pos }
func (n *Percent) Position() int  { return n.Pos }
func (n *Group) Position() int    { return n.Pos }

func (n *Number) String() string {
        return n.Text
}

func (n *Variable) String() string {
        return n.Name
}

// String prints the call as (call name args...). The "call" tag keeps a
// function named group apart from a Group node.
func (n *Call) String() string {
        parts := []string{"call", n.Name}
        for _, arg := range n.Args {
                parts = append(parts, arg.String())
        }
        return "(" + strings.Join(parts, " ") + ")"
}

func (n *Unary) String() string {
        return fmt.Sprintf("(%s %s)", n.Op, n.X)
}

func (n *Binary) String() string {
        return fmt.Sprintf("(%s %s %s)", n.Op, n.L, n.R)
}

func (n *Percent) String() string {
        return fmt.Sprintf("(%% %s)", n.X)
}

func (n *Group) String() string {
        return fmt.Sprintf("(group %s)", n.Inner)
}

package parser

import (
        "fmt"
        "unicode/utf8"
)

// tokenKind identifies what a token is.
type tokenKind int

const (
        tokEOF     tokenKind = iota
        tokNumber            // 12, 3.5, .5
        tokIdent             // sqrt, Ans
        tokPlus              // +
        tokMinus             // -
        tokStar              // *
        tokSlash             // /
        tokCaret             // ^
        tokPercent           // %
        tokLParen            // (
        tokRParen            // )
        tokComma             // ,
)

// symbols maps each single-character token to its kind.
var symbols = map[byte]tokenKind{
        '+': tokPlus,
        '-': tokMinus,
        '*': tokStar,
        '/': tokSlash,
        '^': tokCaret,
        '%': tokPercent,
        '(': tokLParen,
        ')': tokRParen,
        ',': tokComma,
}

// token is one lexical unit of an expression.
type token struct {
        kind tokenKind
        text string // the token as written in the input; empty for tokEOF
        pos  int    // index of the token's first character in the input
}

// lexer splits an expression into tokens on demand: the parser pulls one token
// at a time and stops at the first error, whether lexical or syntactic.
//
// Only ASCII is accepted. The lexer stops at the first non-ASCII character, so
// every position it reports is a byte index, a rune index and a JavaScript
// string index at the same time.
type lexer struct {
        src string
        pos int // index of the next unread byte
}

// next returns the next token, skipping whitespace. At the end of the input it
// returns a tokEOF token positioned at len(src), on every call.
func (l *lexer) next() (token, error) {
        l.skip(isSpace)
        start := l.pos
        if start == len(l.src) {
                return token{kind: tokEOF, pos: start}, nil
        }
        c := l.src[start]
        switch {
        case isDigit(c) || c == '.':
                return l.number()
        case isIdentStart(c):
                l.skip(isIdentPart)
                return token{kind: tokIdent, text: l.src[start:l.pos], pos: start}, nil
        }
        if kind, ok := symbols[c]; ok {
                l.pos++
                return token{kind: kind, text: l.src[start:l.pos], pos: start}, nil
        }
        r, _ := utf8.DecodeRuneInString(l.src[start:])
        return token{}, &SyntaxError{Msg: fmt.Sprintf("unexpected character %q", r), Offset: start}
}

// number scans a literal of the form digits ["." digits] or "." digits.
// Scientific notation is not part of the grammar: in "1e5" the number ends
// after "1" and "e5" is lexed as an identifier, which the parser rejects.
func (l *lexer) number() (token, error) {
        start := l.pos
        l.skip(isDigit)
        if l.peek() == '.' {
                dot := l.pos
                l.pos++
                if l.skip(isDigit) == 0 {
                        return token{}, &SyntaxError{Msg: "decimal point must be followed by a digit", Offset: dot}
                }
        }
        if l.peek() == '.' {
                return token{}, &SyntaxError{Msg: "number has more than one decimal point", Offset: l.pos}
        }
        return token{kind: tokNumber, text: l.src[start:l.pos], pos: start}, nil
}

// peek returns the next unread byte, or 0 at the end of the input.
func (l *lexer) peek() byte {
        if l.pos < len(l.src) {
                return l.src[l.pos]
        }
        return 0
}

// skip advances past the bytes that satisfy ok and reports how many it passed.
func (l *lexer) skip(ok func(byte) bool) int {
        start := l.pos
        for l.pos < len(l.src) && ok(l.src[l.pos]) {
                l.pos++
        }
        return l.pos - start
}

func isSpace(c byte) bool {
        return c == ' ' || c == '\t' || c == '\n' || c == '\r'
}

func isDigit(c byte) bool {
        return '0' <= c && c <= '9'
}

func isIdentStart(c byte) bool {
        return 'a' <= c && c <= 'z' || 'A' <= c && c <= 'Z' || c == '_'
}

func isIdentPart(c byte) bool {
        return isIdentStart(c) || isDigit(c)
}

package parser

import "fmt"

// MaxDepth is the most parentheses, from groups and function calls, that can
// be open at once.
const MaxDepth = 100

// The precedence table. An operator with a higher power binds tighter.
//
//        power  operator  associativity
//        1      + -       left
//        2      * /       left
//        3      unary -   prefix
//        4      ^         right
//        5      %         postfix
var binaryOps = map[tokenKind]struct {
        power      int
        rightAssoc bool
}{
        tokPlus:  {power: 1},
        tokMinus: {power: 1},
        tokStar:  {power: 2},
        tokSlash: {power: 2},
        tokCaret: {power: 4, rightAssoc: true},
}

const (
        // unaryPower sits between * and ^, so -2^2 is -(2^2) while -2*3 is (-2)*3.
        unaryPower = 3
        // percentPower is the tightest, so 2^10% is 2^(10%) and -10% is -(10%).
        percentPower = 5
)

// Parse turns an expression into a syntax tree. Malformed input returns a
// *SyntaxError, and parentheses nested deeper than MaxDepth return a
// *NestingError. Both report the position of the offending character.
func Parse(src string) (Node, error) {
        p := &parser{lex: lexer{src: src}}
        if err := p.advance(); err != nil {
                return nil, err
        }
        if p.tok.kind == tokEOF {
                return nil, &SyntaxError{Msg: "empty expression", Offset: p.tok.pos}
        }
        n, err := p.expr(0)
        if err != nil {
                return nil, err
        }
        if p.tok.kind != tokEOF {
                return nil, p.unexpected()
        }
        return n, nil
}

type parser struct {
        lex   lexer
        tok   token // the current token, not yet consumed
        depth int   // number of open parentheses around the current token
}

// advance replaces the current token with the next one from the lexer.
func (p *parser) advance() error {
        tok, err := p.lex.next()
        if err != nil {
                return err
        }
        p.tok = tok
        return nil
}

// expr parses an expression whose operators, outside parentheses, all have at
// least minPower. A looser operator ends the expression and is left for the
// caller.
func (p *parser) expr(minPower int) (Node, error) {
        left, err := p.unary()
        if err != nil {
                return nil, err
        }
        for {
                if p.tok.kind == tokPercent && percentPower >= minPower {
                        left = &Percent{X: left, Pos: p.tok.pos}
                        if err := p.advance(); err != nil {
                                return nil, err
                        }
                        continue
                }
                op, ok := binaryOps[p.tok.kind]
                if !ok || op.power < minPower {
                        return left, nil
                }
                opTok := p.tok
                if err := p.advance(); err != nil {
                        return nil, err
                }
                // The right operand of a left-associative operator only takes tighter
                // operators, so 1-2-3 is (1-2)-3. A right-associative operator also
                // takes its own level, so 2^3^2 is 2^(3^2).
                next := op.power + 1
                if op.rightAssoc {
                        next = op.power
                }
                right, err := p.expr(next)
                if err != nil {
                        return nil, err
                }
                left = &Binary{Op: opTok.text, L: left, R: right, Pos: opTok.pos}
        }
}

// unary parses an operand with any number of leading minus signs. The operand
// of a minus sign only takes operators that bind tighter than unaryPower.
func (p *parser) unary() (Node, error) {
        if p.tok.kind != tokMinus {
                return p.primary()
        }
        opTok := p.tok
        if err := p.advance(); err != nil {
                return nil, err
        }
        x, err := p.expr(unaryPower)
        if err != nil {
                return nil, err
        }
        return &Unary{Op: opTok.text, X: x, Pos: opTok.pos}, nil
}

// primary parses a number, a variable, a function call or a parenthesized
// expression.
func (p *parser) primary() (Node, error) {
        switch p.tok.kind {
        case tokNumber:
                n := &Number{Text: p.tok.text, Pos: p.tok.pos}
                if err := p.advance(); err != nil {
                        return nil, err
                }
                return n, nil
        case tokIdent:
                name := p.tok
                if err := p.advance(); err != nil {
                        return nil, err
                }
                if p.tok.kind == tokLParen {
                        return p.call(name)
                }
                return &Variable{Name: name.text, Pos: name.pos}, nil
        case tokLParen:
                return p.group()
        }
        return nil, p.unexpected()
}

// call parses the argument list that follows the function name. Any name is
// accepted: the evaluator looks it up, so adding a function does not touch the
// parser.
func (p *parser) call(name token) (Node, error) {
        open, err := p.openParen()
        if err != nil {
                return nil, err
        }
        var args []Node
        if p.tok.kind != tokRParen {
                for {
                        arg, err := p.expr(0)
                        if err != nil {
                                return nil, err
                        }
                        args = append(args, arg)
                        if p.tok.kind != tokComma {
                                break
                        }
                        if err := p.advance(); err != nil {
                                return nil, err
                        }
                }
        }
        if err := p.closeParen(open); err != nil {
                return nil, err
        }
        return &Call{Name: name.text, Args: args, Pos: name.pos}, nil
}

// group parses "(" expression ")" into a Group node.
func (p *parser) group() (Node, error) {
        open, err := p.openParen()
        if err != nil {
                return nil, err
        }
        inner, err := p.expr(0)
        if err != nil {
                return nil, err
        }
        if err := p.closeParen(open); err != nil {
                return nil, err
        }
        return &Group{Inner: inner, Pos: open.pos}, nil
}

// openParen consumes "(" and enforces MaxDepth. It returns the "(" token for
// the matching closeParen.
func (p *parser) openParen() (token, error) {
        open := p.tok
        if p.depth == MaxDepth {
                return token{}, &NestingError{Offset: open.pos}
        }
        p.depth++
        if err := p.advance(); err != nil {
                return token{}, err
        }
        return open, nil
}

// closeParen consumes the ")" that matches open.
func (p *parser) closeParen(open token) error {
        if p.tok.kind != tokRParen {
                if p.tok.kind == tokEOF {
                        return &SyntaxError{Msg: "unclosed parenthesis", Offset: open.pos}
                }
                return p.unexpected()
        }
        p.depth--
        return p.advance()
}

// unexpected reports the current token as a syntax error.
func (p *parser) unexpected() error {
        if p.tok.kind == tokEOF {
                return &SyntaxError{Msg: "unexpected end of expression", Offset: p.tok.pos}
        }
        return &SyntaxError{Msg: fmt.Sprintf("unexpected '%s'", p.tok.text), Offset: p.tok.pos}
}

// file is a file of the build, read into memory with its response headers.
type file struct {
        data         []byte
        contentType  string
        cacheControl string
        etag         string
}

// NewHandler serves the files of the frontend build fsys with GET and HEAD.
// "/" serves index.html, and every other path serves the file of that name.
// Directories are not listed, files and directories whose names start with a
// dot are not served, and other methods get 405.
//
// The files are read once, here. A file with an extension missing from
// contentTypes is an error, so it fails at startup instead of being served
// with a guessed type.
func NewHandler(fsys fs.FS) (http.Handler, error) {
        files := make(map[string]file)
        err := fs.WalkDir(fsys, ".", func(name string, d fs.DirEntry, err error) error {
                if err != nil {
                        return err
                }
                if name != "." && strings.HasPrefix(d.Name(), ".") {
                        if d.IsDir() {
                                return fs.SkipDir
                        }
                        return nil
                }
                if d.IsDir() {
                        return nil
                }
                f, err := readFile(fsys, name)
                if err != nil {
                        return err
                }
                files["/"+name] = f
                return nil
        })
        if err != nil {
                return nil, fmt.Errorf("web: %w", err)
        }
        if index, ok := files["/index.html"]; ok {
                files["/"] = index
        }

        mux := http.NewServeMux()
        mux.HandleFunc("GET /", func(w http.ResponseWriter, r *http.Request) {
                f, ok := files[r.URL.Path]
                if !ok {
                        http.NotFound(w, r)
                        return
                }
                h := w.Header()
                h.Set("Content-Type", f.contentType)
                h.Set("Cache-Control", f.cacheControl)
                h.Set("ETag", f.etag)
                // ServeContent answers If-None-Match with 304 and handles HEAD and
                // Range requests. Embedded files have no modification time.
                http.ServeContent(w, r, "", time.Time{}, bytes.NewReader(f.data))
        })
        return mux, nil
}

func readFile(fsys fs.FS, name string) (file, error) {
        contentType, ok := contentTypes[path.Ext(name)]
        if !ok {
                return file{}, fmt.Errorf("%s: no content type for its extension", name)
        }
        data, err := fs.ReadFile(fsys, name)
        if err != nil {
                return file{}, err
        }
        cacheControl := cacheRevalidate
        if strings.HasPrefix(name, "assets/") {
                cacheControl = cacheImmutable
        }
        sum := sha256.Sum256(data)
        return file{
                data:         data,
                contentType:  contentType,
                cacheControl: cacheControl,
                etag:         `"` + hex.EncodeToString(sum[:16]) + `"`,
        }, nil
}
